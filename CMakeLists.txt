cmake_minimum_required(VERSION 3.16)
project(pi4_radio_ui C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT LVGL_DIR)
  set(LVGL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lvgl)
endif()
set(LV_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_DIR} ${CMAKE_CURRENT_BINARY_DIR}/lvgl)
add_executable(radio-ui src/main.c src/mpd_client.c src/ui_pi.c src/backlight_stub.c src/cover.c)
target_include_directories(radio-ui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(radio-ui PRIVATE -Wall -Wextra)
target_link_libraries(radio-ui PRIVATE lvgl mpdclient png jpeg m)
